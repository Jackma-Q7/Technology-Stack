## 4.2-C与⾯向对象

&emsp;&emsp;C语⾔是⾯向过程的语⾔，但我们可以借助C++/JAVA⾯向对象思想，实现类似"继承"、"多 态"、"封装"、"重载"的功能。从⽽服务于架构设计。  

### 4.2.1 C与"继承"  
&emsp;&emsp;继承是使⽤已存在的类作为基础建⽴新的类的技术。新的类可以增加新的数据类型或者⽅法，也 可以使⽤⽗类的功能，但不能选择性的继承⽗类。通过继承可以很好的复⽤以前的代码，提⾼开发效 率。  

```cpp
 #include <iostream>

 using namespace std;

 // 基类
 class shape  {
  protected:
    int width;
    int height;

  public:
    void setWidth(int w) {
      width = w;
    }

    void setHeight(int h) {
      height = h;
    }
 };

// 派⽣类
 class rectangle : public shape
 {
  public:
    int getArea() { 
      return (width * height); 
    }
 };

 int main(void)
 {
  rectangle rect;
  rect.setWidth(5);
  rect.setHeight(7);

  // 输出对象的⾯积
  cout << "Total area: " << rect.getArea() << endl;
  
  return 0;
 }
```

```c
#include <stdio.h>
#include <string.h>

// 基类
struct shape {
  int width;
  int height;
  void (*setWidth)(struct shape*s, int w);
  void (*setHeight)(struct shape*s, int h);
};

void setWidth(struct shape*s, int w)
{
  s->width = w;
}

void setHeight(struct shape*s, int h)
{
  s->height = h;
}

// 派⽣类
struct rectangle {
  struct shape s;
  int (*getArea)(struct shape *s);
};

int getArea(struct shape *s)
{
  return (s->width * s->height);
}

int main(void)
{
  struct rectangle rect = {
    .s.setWidth = setWidth,
    .s.setHeight = setHeight,
    .getArea = getArea,
  };

  rect.s.setWidth(&rect.s, 5);
  rect.s.setHeight(&rect.s, 7);
  // 输出对象的⾯积
 
  printf("Total area: %d \n", rect.getArea(&rect.s));
  return 0;
}
```

可以想象成：

```plain
rect
 ├── s
 │    ├── width = 0   （未显式初始化时，聚合初始化里没写的成员会补 0）
 │    ├── height = 0
 │    ├── setWidth  -> 指向函数 setWidth
 │    └── setHeight -> 指向函数 setHeight
 └── getArea -> 指向函数 getArea
```

这里 `width` 和 `height` 没写初始化值，所以会被补成 0。

可以把整个过程画成这样：

```plain
main 开始
  ↓
创建 rect，并初始化函数指针
  ↓
rect.s.setWidth(&rect.s, 5)
  ↓
跳到 setWidth
  ↓
rect.s.width = 5
  ↓
返回 main
  ↓
rect.s.setHeight(&rect.s, 7)
  ↓
跳到 setHeight
  ↓
rect.s.height = 7
  ↓
返回 main
  ↓
rect.getArea(&rect.s)
  ↓
跳到 getArea
  ↓
返回 5 * 7 = 35
  ↓
返回 main
  ↓
printf 输出 Total area: 35
  ↓
return 0
  ↓
程序结束
```

###  4.2.3C与"多态"
&emsp;&emsp;多态的概念是指：通过⽗类的指针来调⽤⼦类中的⽅法。作⽤是把不同的⼦类对象当做⽗类来看，屏蔽不同⼦类之间的差异，⽤于适应需求的不断变化，本质上是⾯向抽象编程，在C++中，多态的

实现是通过虚函数来实现。体会⼀下以下这个例⼦：

```cpp
#include <stdio.h>
#include <string.h>

class person {
protected:
    char name[24];
    int age;

public:
    virtual void get_name(char *n) {
        strcpy(n, name);
    }

    person(const char *n, int a) {
        strcpy(name, n);
        age = a;
    }

    virtual ~person() {}
};

class usa_person : public person {
public:
    void get_name(char *n) override {
        sprintf(n, "%s", name);
        sprintf(n + strlen(n), "%s", "-usa");
    }

    usa_person(const char *n, int a) : person(n, a) {
    }
};

class china_person : public person {
public:
    void get_name(char *n) override {
        sprintf(n, "%s", name);
        sprintf(n + strlen(n), "%s", "-china");
    }

    china_person(const char *n, int a) : person(n, a) {
    }
};

int main(void)
{
    char name[24] = {0};

    person *p = new usa_person("xiaohua", 21);
    p->get_name(name);
    printf("name: %s\n", name);
    delete p;

    p = new china_person("xiaohua", 21);
    p->get_name(name);
    printf("name: %s\n", name);
    delete p;

    return 0;
}
```

###  4.2.4 C与"重载"  
&emsp;&emsp;重载的概念是指：在同⼀个范围中声明⼏个同名函数，但同名函数的形参不同（个数、类型、顺 序），⼀般是⽤来处理功能类似但是数据类型不同的问题。严格意义上C的编译器不⽀持重载，但我们有⼀些⽅法达到相同的⽬的  

```c
//printf函数是C中典型有重载效果的函数
 
 #include <stdio.h>

  printf("%s %s \n", "hello world 1", "hello world 2");

  printf("%s %d \n", "hello world 1", 10);

  return 0;
 }
```

```c
 #include <stdio.h>
 #include <stdarg.h>

 //这⾥约定n为后⾯参数的个数（⼀种规则）
 static void va_func(int n, ...)
 {
  va_list ptr;
  va_start(ptr, n);
  while (n--) {
    printf("%d \n", va_arg(ptr, int));
  }
  va_end(ptr);
 }

 int main(void)
 {
  va_func(1, 100);
  va_func(2, 200, 300);
  return 0;
 }
```

#### 4.2.4.2 回调函数+void*
```Cpp
 #include <stdio.h>

 class swap {
    public:
      void exchange(int *a, int *b) {
      int tmp = *a;
      *a = *b;
      *b = tmp;
    }
    void exchange(double *a, double *b) {
      double tmp = *a;
      *a = *b;
      *b = tmp;
    }
 };

 int main(void)
 {
  swap s;
  int a = 10;
  int b = 20;
  double c = 1.23;
  double d = 4.56;

  s.exchange(&a, &b);
  s.exchange(&c, &d);
  
  printf("a = %d, b = %d \n", a , b);
  printf("c = %f, d = %f \n", c , d);
  return 0;
 }
```

```c
//实现⼀个可以交换不同数据类型的swap函数
 #include <stdio.h>
 #include <stdarg.h>

 typedef void (*func)(void*, void*);
 void swap(func exchange, void *a, void *b)
 {
  exchange(a, b);
 }

 void exchange_int(void *a, void *b)
 {
  int *_a = (int*)a;
  int *_b = (int*)b;
  int tmp = 0;
  tmp = *_a;
  *_a = *_b;
  *_b= tmp ;
 }

 void exchange_double(void *a, void *b)
 {
  double*_a = (double*)a;
  double*_b = (double*)b;
  double tmp = 0;
  tmp = *_a;
  *_a = *_b;
  *_b= tmp;
 }

 void main(void)
 {
  int a = 10;
  int b = 20;
  double c = 1.23;
  double d = 4.56;

  swap(exchange_int, &a, &b);
  swap(exchange_double, &c, &d);

  printf("a = %d, b = %d \n", a , b);
  printf("c = %f, d = %f \n", c , d);
 }

//运⾏结果: 
a = 20, b = 10 
c = 4.560000, d = 1.230000
```
