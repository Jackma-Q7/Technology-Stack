## **6.2.6贴合地面，添加虚拟部件**
**base.urdf.xacro**

```xml
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro">

  <!-- base_xacro：定义机器人底盘的基础 TF -->
  <xacro:macro name="base_xacro" params="length radius">

    <!-- 机器人最底部的坐标系（一般用于导航） -->
    <link name="base_footprint"/>

    <!-- base_footprint → base_link 的固定连接 -->
    <joint name="base_joint" type="fixed">
      <parent link="base_footprint"/>
      <child  link="base_link"/>

      <!--
        将 base_link 提升一定高度（机器人体形高度的一部分）
        计算式：
          length/2.0 + 0.032 - 0.001
        其中：
          length/2.0 = 高度中心值
          0.032 = 车轮半径或底盘高度
          0.001 = 微调
      -->
      <origin xyz="0.0 0.0 ${length/2.0 + 0.032 - 0.001}" rpy="0 0 0"/>
    </joint>

    <!-- 你还可以在这里添加 base_link 的几何可视化，比如 cylinder 或 box -->

  </xacro:macro>

</robot>
```
