## **6.3.1为机器人部件添加碰撞属性**
```xml
<?xml version="1.0"?>
<robot xmlns:xacro="http://www.ros.org/wiki/xacro">

  <!-- base_xacro：机器人主体底盘几何 -->
  <xacro:macro name="base_xacro" params="length radius">

    <link name="base_link">

      <!-- 可视化模型（RViz 看到的外形） -->
      <visual>
        <!-- 在自身坐标系下的偏移 -->
        <origin xyz="0 0 0" rpy="0 0 0"/>
        <geometry>
          <!-- 可视化圆柱体 -->
          <cylinder length="${length}" radius="${radius}"/>
        </geometry>
        <material name="white">
          <color rgba="1.0 1.0 1.0 0.5"/>
        </material>
      </visual>

      <!-- 碰撞模型（用于物理仿真、碰撞检测） -->
      <collision>
        <!-- 一般 collision 与 visual 保持一致的原点和形状 -->
        <origin xyz="0 0 0" rpy="0 0 0"/>
        <geometry>
          <!-- 与 visual 相同的形状 -->
          <cylinder length="${length}" radius="${radius}"/>
        </geometry>
        <material name="white">
          <color rgba="1.0 1.0 1.0 0.5"/>
        </material>
      </collision>

    </link>

  </xacro:macro>

</robot>

```

给其他的一样加入<collision>
