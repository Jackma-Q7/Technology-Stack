```cpp
#include <Arduino.h>
#include <Esp32PcntEncoder.h>

// 创建一个数组，用于存储两个编码器对象
Esp32PcntEncoder encoders[2];

void setup()
{
    // 1. 初始化串口通信
    Serial.begin(115200);   // 设置串口波特率为 115200

    // 2. 初始化编码器
    // 编码器 0：GPIO 32 (A 相)，GPIO 33 (B 相)
    encoders[0].init(0, 32, 33);

    // 编码器 1：GPIO 26 (A 相)，GPIO 25 (B 相)
    encoders[1].init(1, 26, 25);
}

void loop()
{
    delay(10);  // 等待 10 ms

    // 读取并打印两个编码器的计数值
    Serial.printf(
        "tick1=%d, tick2=%d\n",
        encoders[0].getTicks(),
        encoders[1].getTicks()
    );
}
```

```cpp
#include <Arduino.h>
#include <Esp32PcntEncoder.h>
#include <Esp32McpwmMotor.h>

// 电机与编码器对象
Esp32McpwmMotor motor;
Esp32PcntEncoder encoders[2];

// 编码器与速度相关变量
int64_t last_ticks[2];        // 上一次编码器计数值
int32_t delta_ticks[2];       // 两次读取之间的差值
int64_t last_update_time;     // 上一次更新时间（ms）
float current_speeds[2];      // 当前电机速度（m/s）

void setup()
{
    Serial.begin(115200);

    // 初始化编码器
    encoders[0].init(0, 32, 33);   // 编码器 0：GPIO32 / GPIO33
    encoders[1].init(1, 26, 25);   // 编码器 1：GPIO26 / GPIO25

    // 初始化电机（PWM 引脚）
    motor.attachMotor(0, 22, 23);  // 电机 0：GPIO22 / GPIO23
    motor.attachMotor(1, 12, 13);  // 电机 1：GPIO12 / GPIO13

    // 设置初始电机速度（占空比 70%）
    motor.updateMotorSpeed(0, 70);
    motor.updateMotorSpeed(1, 70);

    // 初始化时间与编码器计数
    last_update_time = millis();
    last_ticks[0] = encoders[0].getTicks();
    last_ticks[1] = encoders[1].getTicks();
}

void loop()
{
    delay(10);   // 控制周期 10 ms

    // 计算时间差（ms）
    uint64_t dt = millis() - last_update_time;
    if (dt == 0) return;

    // 计算编码器增量
    delta_ticks[0] = encoders[0].getTicks() - last_ticks[0];
    delta_ticks[1] = encoders[1].getTicks() - last_ticks[1];

    // 根据编码器增量计算速度
    // 0.1051566：由轮子半径、编码器分辨率换算得到
    current_speeds[0] = float(delta_ticks[0]) * 0.1051566f / dt;
    current_speeds[1] = float(delta_ticks[1]) * 0.1051566f / dt;

    // 更新历史数据
    last_update_time = millis();
    last_ticks[0] = encoders[0].getTicks();
    last_ticks[1] = encoders[1].getTicks();

    // 输出当前速度
    Serial.printf(
        "speed1=%f m/s, speed2=%f m/s\n",
        current_speeds[0],
        current_speeds[1]
    );
}
```
