## 5.2.3通过Python查询TF关系
```python
import rclpy
from rclpy.node import Node
from tf2_ros import TransformListener, Buffer
from tf_transformations import euler_from_quaternion

class TFListener(Node):

    def __init__(self):
        super().__init__('tf2_listener')
        self.buffer_ = Buffer()
        self.listener_ = TransformListener(self.buffer_, self)
        # 创建定时器，每 1 秒调用一次 get_transform 函数
        self.timer_ = self.create_timer(1, self.get_transform)

    def get_transform(self):
        try:
            # 查询 base_link → bottle_link 的坐标变换
            result = self.buffer_.lookup_transform(
                'base_link', 'bottle_link',
                rclpy.time.Time(seconds=0),
                rclpy.time.Duration(seconds=1)#######类型
            )
            transform = result.transform

            # 四元数转欧拉角
            rotation_euler = euler_from_quaternion([
                transform.rotation.x,
                transform.rotation.y,
                transform.rotation.z,
                transform.rotation.w
            ])#数组

            # 打印平移和旋转信息
            self.get_logger().info(
                f'平移: {transform.translation}, 旋转四元数: {transform.rotation}; '
                f'旋转欧拉角: {1}'
            )

        except Exception as e:
            # 查询失败时打印错误
            self.get_logger().warn(
                f'不能够获取坐标变换, 原因: {str(e)}'
            )


def main():
    rclpy.init()
    node = TFListener()
    rclpy.spin(node)
    rclpy.shutdown()

```
