```cpp
#include "chapt4_interfaces/srv/patrol.hpp"

        "patrol",//服务名字
        [&](const std::shared_ptr<Patrol::Request> request,//回调函数
        std::shared_ptr<Patrol::Response> response) -> void {
            if ((0 < request->target_x && request->target_x < 12.0f)
                && (0 < request->target_y && request->target_y < 12.0f)) {
                target_x_ = request->target_x;
                target_y_ = request->target_y;
                response->result = Patrol::Response::SUCCESS;
            } else {
                response->result = Patrol::Response::FAIL;
            }
        });
}

private:
rclcpp::Service<Patrol>::SharedPtr patrol_server_;
};
```

```cpp
#include "rclcpp/rclcpp.hpp"                 // ROS2 C++ 客户端库
#include "geometry_msgs/msg/twist.hpp"       // 速度消息类型 Twist
#include <chrono>                            // 使用 chrono 进行时间处理
#include"turtlesim/msg/Pose.hpp"
#include "chapt4_interfaces/srv/patrol.hpp"
using Patrol = chapt4_interfaces::srv::Patrol;
using namespace std::chrono_literals;

class Turtlecontrol : public rclcpp::Node
{
private: 
    rclcpp::Service<Patrol>::SharedPtr patrol_server_;//////////////////////////
    rclcpp::Subscription<turtlesim::msg::Pose>::SharedPtr subcriber_;
    // 发布者指针：发布 Twist 消息给小乌龟
    rclcpp::Publisher<geometry_msgs::msg::Twist>::SharedPtr publisher_;

    double target_x{1.0};// 目标位置x，设置默认值 1.0
    double target_y{1.0};// 目标位置y，设置默认值 1.0
    double k {1.0};//比例系数，控制输出=误差x比例系数
    double max_speed_{3.0};//最大线速度，设置默认值3.0

public:
    // 构造函数（node_name 为节点名称）
    explicit Turtlecontrol(const std::string& node_name)
        : Node(node_name)      // 调用父类 Node 的构造函数
    {

        TurtleController() : Node("turtle_controller") 
        {
        patrol_server_ = this->create_service<Patrol>(
            "patrol",//服务名字
            [&](const std::shared_ptr<Patrol::Request> request,//回调函数
            std::shared_ptr<Patrol::Response> response) -> void {
                if ((0 < request->target_x && request->target_x < 12.0f)
                    && (0 < request->target_y && request->target_y < 12.0f)) {
                    target_x_ = request->target_x;
                    target_y_ = request->target_y;
                    response->result = Patrol::Response::SUCCESS;
                } else {
                    response->result = Patrol::Response::FAIL;
                }
            });
            
        // 创建一个 Twist 消息发布者，话题名称 "/turtle1/cmd_vel"，队列长度 10
        publisher_ = this->create_publisher<geometry_msgs::msg::Twist>
            ("/turtle1/cmd_vel", 10);

        subscriber_ = this ->create_subscriber<turtlesim::msg::Pose>
            ("/turtle1/pose",10,std::bind(&Turtlecontrol::on_pose_received_,
            this,std::placeholder::_1))
        }

private:
    // 定时器回调函数，每 1 秒执行一次
    void on_pose_received_(const turtlesim::msg::Pose::SharedPtr pose)
    {
        auto msg = geometry_msgs::msg::Twist();   // 创建速度消息对象

        //1.记录当前位置
        double current_x=pose->x;
        double current_y=pose->y;
        RCLCPP_INFO(this->get_logger(),"当前位置:(x=f,y=名f)",current_x,current_y);
        //2.计算与目标之间的距离，以及与当前海龟朝向的角度差

        double distance =std::sqrt(
        (target_x-current_x)*(target_x-current_x)+
        (target_y-current_y)*(target_y-current_y)
        );
        double angle=std::atan2(target_y_-currenty,target_x_-current_x)-pose->theta;

        //3.控制策略:距离大于0.1继续运动，角度差大于0.2则原地旋转，否则直行
        if(distance>0.1)
        {
            if(fabs(angle)>0.2)
            {
                message.angular.z=fabs(angle);
            }
            else
            {
            //通过比例控制器计算输出速度
            message.linear.x=k*distance;
            }
        }
        
    //4.限制最大值并发布消息
        if(message.linear.x>max_speed_)
        {
        message.linear.x=max_speed_;
        }
        
        publisher_->publish(msg);   // 发布消息
    }
};

int main(int argc, char * argv[])
{
    rclcpp::init(argc, argv);   // 初始化 ROS2

    auto node = std::make_shared<Turtlecontrol>("turtle_control");

    rclcpp::spin(node);         // 进入循环，等待并执行回调
    rclcpp::shutdown();         // 关闭 ROS2

    return 0;
}
```
