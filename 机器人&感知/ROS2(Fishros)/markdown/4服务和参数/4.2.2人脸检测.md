代码

```python
import face_recognition  # 人脸检测库
import cv2  # OpenCV（图像处理）
from ament_index_python.packages import get_package_share_directory  # ROS 2功能包路径工具

def main():
    # 1. 获取ROS 2功能包内的图片路径
    default_image_path = get_package_share_directory('demo_python_service') 
    + '/resource/default.jpg'
    
    # 2. 用OpenCV加载图片
    image = cv2.imread(default_image_path)
    
    # 3. 检测图片中的所有人脸（使用HOG模型，上采样1次）
    face_locations = face_recognition.face_locations(
        image, number_of_times_to_upsample=1, model='hog'
    )
    
    # 4. 给每个人脸绘制蓝色边框（颜色(255,0,0)，线宽4）
    for top, right, bottom, left in face_locations:
        cv2.rectangle(image, (left, top), (right, bottom), (255, 0, 0), 4)
    
    # 5. 显示检测结果窗口，按任意键关闭
    cv2.imshow('Face Detection', image)
    cv2.waitKey(0)
```
