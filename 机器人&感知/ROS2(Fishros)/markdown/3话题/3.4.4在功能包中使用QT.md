## 3.4.4在功能包中使用QT
```cpp
#include <QApplication>
#include <QLabel>
#include <QString>

int main(int argc, char* argv[]) {
    // 创建Qt应用程序对象：argc/argv是命令行参数，是Qt程序的必要初始化
    QApplication app(argc, argv);
    
    // 动态创建QLabel标签控件（用于显示文本）
    QLabel* label = new QLabel();

    // 将C++标准字符串转换为Qt的QString类型
    QString message = QString::fromStdString("Hello Qt!");

    // 设置标签的显示文本为message
    label->setText(message);

    // 显示标签控件（默认隐藏，需手动调用show()显示）
    label->show();

    // 启动Qt的事件循环：程序进入等待用户操作的状态（如窗口关闭）
    app.exec();

    return 0;
}
```

**CMakeList.txt**

```cmake
# 1. 指定最低CMake版本要求（确保构建工具兼容性）
cmake_minimum_required(VERSION 3.8)

# 2. 查找依赖包（REQUIRED表示找不到则构建失败）
# 查找ROS 2自定义接口包（status_interfaces，用于消息/服务通信）
find_package(status_interfaces REQUIRED)
# 查找Qt5库，明确依赖Widgets组件（Qt GUI核心控件，如窗口、标签等）
find_package(Qt5 REQUIRED COMPONENTS Widgets)

# 3. 编译可执行文件
# 将src/hello_qt.cpp源文件编译为名为hello_qt的可执行程序
add_executable(hello_qt src/hello_qt.cpp)

# 4. 链接依赖库
# 将hello_qt程序与Qt5::Widgets库链接，保证Qt GUI功能可调用
target_link_libraries(hello_qt Qt5::Widgets)

# 5. 安装配置（符合ROS 2标准安装路径）
# 将编译后的hello_qt程序安装到 lib/[项目名] 目录下
install(TARGETS hello_qt
  DESTINATION lib/${PROJECT_NAME})

# 6. ROS 2包标识
# 生成ROS 2所需的包配置文件，标记为ROS 2包
ament_package()
```
