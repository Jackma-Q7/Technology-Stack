## 3.3.2订阅pose实现闭环控制
```cpp
#include "rclcpp/rclcpp.hpp"                 // ROS2 C++ 客户端库
#include "geometry_msgs/msg/twist.hpp"       // 速度消息类型 Twist
#include "turtlesim/msg/pose.hpp"            // turtlesim 位姿消息

#include <chrono>
#include <cmath>
#include <functional>

using namespace std::chrono_literals;

class Turtlecontrol : public rclcpp::Node
{
private:
    // 订阅者指针：订阅小乌龟当前位置
    rclcpp::Subscription<turtlesim::msg::Pose>::SharedPtr subscriber_;

    // 发布者指针：发布 Twist 消息给小乌龟
    rclcpp::Publisher<geometry_msgs::msg::Twist>::SharedPtr publisher_;

    double target_x_{1.0};      // 目标位置 x
    double target_y_{1.0};      // 目标位置 y
    double k_{1.0};             // 线速度比例系数
    double max_speed_{3.0};     // 最大线速度

public:
    explicit Turtlecontrol(const std::string& node_name)
        : Node(node_name)
    {
        publisher_ = this->create_publisher<geometry_msgs::msg::Twist>(
            "/turtle1/cmd_vel", 10
        );

        subscriber_ = this->create_subscription<turtlesim::msg::Pose>(
            "/turtle1/pose",
            10,
            std::bind(&Turtlecontrol::on_pose_received_, this, std::placeholders::_1)
        );
    }

private:
    void on_pose_received_(const turtlesim::msg::Pose::SharedPtr pose)
    {
        auto msg = geometry_msgs::msg::Twist();

        // 1. 记录当前位置
        double current_x = pose->x;
        double current_y = pose->y;

        RCLCPP_INFO(
            this->get_logger(),
            "当前位置: x=%.2f, y=%.2f",
            current_x,
            current_y
        );

        // 2. 计算与目标之间的距离
        double distance = std::sqrt(
            (target_x_ - current_x) * (target_x_ - current_x) +
            (target_y_ - current_y) * (target_y_ - current_y)
        );

        // 目标方向角
        double target_angle = std::atan2(
            target_y_ - current_y,
            target_x_ - current_x
        );

        // 当前朝向与目标方向之间的角度差
        double angle = target_angle - pose->theta;

        // 角度归一化到 [-pi, pi]
        while (angle > M_PI) angle -= 2.0 * M_PI;
        while (angle < -M_PI) angle += 2.0 * M_PI;

        // 3. 控制策略
        if (distance > 0.1)
        {
            if (std::fabs(angle) > 0.2)
            {
                // 角度偏差较大，先旋转
                msg.angular.z = angle;
            }
            else
            {
                // 朝向基本正确，向前走
                msg.linear.x = k_ * distance;
            }
        }
        else
        {
            // 到达目标点，停止
            msg.linear.x = 0.0;
            msg.angular.z = 0.0;
        }

        // 4. 限制最大线速度
        if (msg.linear.x > max_speed_)
        {
            msg.linear.x = max_speed_;
        }

        publisher_->publish(msg);
    }
};

int main(int argc, char * argv[])
{
    rclcpp::init(argc, argv);

    auto node = std::make_shared<Turtlecontrol>("turtle_control");

    rclcpp::spin(node);

    rclcpp::shutdown();

    return 0;
}
```
