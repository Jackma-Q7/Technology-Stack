## 3.2.1通过话题发布小说
鱼老板需求:下载小说并通过话题间隔5s发布一行核心问题

问题1:怎么下载小说?request

问题2:怎么发布?确定名字和接口

问题3:怎么间隔5s发布?Timer定时器

1.创建功能包

```
ros2 pkg create demo_python_topic --build-type ament_pyhton --dependencies rclpy exmple_interfaces --license Apache-2.0
```
2.写代码

```python
import rclpy
from rclpy.node import Node
import requests
from example_interfaces.msg import String
from queue import Queue

class NovelPubNode(Node):

    def __init__(self, node_name):
        super().__init__(node_name)
        self.novels_queue = Queue()  # 创建队列，存放小说

        # 创建话题发布者，发布小说                     #消息接口
        self.novel_publisher = self.create_publisher(String, 'novel', 10)
        //消息接口，话题名字，qos_profile
        # 创建定时器，每 5 秒发布一行
        self.timer = self.create_timer(5, self.timer_callback)

    def download_novel(self, url):
        response = requests.get(url)
        response.encoding = 'utf-8'
        self.get_logger().info(f'下载完成: {url}')

        # 按行分割，放入队列
        for line in response.text.splitlines():
            self.novels_queue.put(line)

    def timer_callback(self):
        # 队列中有数据，取出一行并发布
        if self.novels_queue.qsize() > 0:
            msg = String()  # 实例化一个消息
            msg.data = self.novels_queue.get()  # 赋值
            self.novel_publisher.publish(msg)  # 发布消息
            self.get_logger().info(f'发布了一行小说: {msg.data}')
        else:
            self.get_logger().info('队列为空，等待下载内容...')


def main():
    rclpy.init()
    node = NovelPubNode('novel_pub')

    # 下载小说
    node.download_novel('http://localhost:8000/novel1.txt')

    rclpy.spin(node)
    rclpy.shutdown()
```
